============================================================
  Git 操作演示 -- 智能家居项目版本管理示例
============================================================

[环境] 假设项目根目录为 d:\smart-home-esp32\
        已安装 Git，且配好用户名和邮箱


============================================================
第一步: 首次配置 (只需做一次)
============================================================

git config --global user.name   "你的名字"
git config --global user.email  "你的邮箱@example.com"


============================================================
第二步: 初始化仓库 & 首次提交
============================================================

cd d:\smart-home-esp32

git init                              # 在当前目录创建 .git 仓库

git add .                             # 暂存所有文件 (会读 .gitignore 跳过忽略项)

git commit -m "init: 项目初始化，搭建骨架"     # 首次提交


============================================================
第三步: 日常开发流水 (做完一个功能就提交一次)
============================================================

# --- 写完温湿度传感器驱动 ---
git add src/sensors/dht_sensor.h
git commit -m "feat: 添加 DHT11 温湿度传感器驱动"

# --- 写完 LED 控制 ---
git add src/actuators/led_control.*
git commit -m "feat: 添加 LED 灯控制模块，支持 PWM 调光"

# --- 改了一堆东西，想全部暂存 ---
git add .
git commit -m "feat: 完成 MQTT 通信模块与联动规则引擎"


============================================================
第四步: 查看状态 / 日志 / 差异 (排查问题用)
============================================================

git status                     # 看哪些文件改过还没提交
git log --oneline              # 看提交历史 (一行一条，简洁)
git diff                       # 看还没暂存的具体改动
git diff --staged              # 看已暂存、还没提交的改动


============================================================
第五步: 写错了想回退
============================================================

# 还没 add，撤销工作区的改动
git checkout -- src/main.cpp

# 已经 add 了，从暂存区撤回来
git reset HEAD src/main.cpp

# 已经 commit 了，回退到上一个提交 (保留代码改动)
git reset --soft HEAD~1

# 已经 commit 了，彻底丢掉 (代码也丢掉，慎用！)
git reset --hard HEAD~1


============================================================
第六步: 关联远程仓库 (GitHub / Gitee)
============================================================

# 先在 GitHub 上新建一个空仓库 (不要勾选 README)，然后：

git remote add origin https://github.com/你的用户名/smart-home-esp32.git

git branch -M main                  # 主分支改名为 main

git push -u origin main             # 推送本地代码到远程


============================================================
第七步: 后续推送 & 拉取
============================================================

git push                            # 推送本地新提交到远程
git pull                            # 拉取远程更新到本地

# 换台电脑继续开发:
git clone https://github.com/你的用户名/smart-home-esp32.git


============================================================
推荐提交信息前缀 (一目了然)
============================================================

feat:     新功能       (feat: 添加 MQTT 断线重连)
fix:      修 bug       (fix: 修复 PIR 传感器误触发)
refactor: 重构代码     (refactor: 将延时函数改为定时器)
docs:     文档         (docs: 更新硬件接线图)
chore:    杂项配置     (chore: 更新 .gitignore)


============================================================
一条龙演示 (按顺序敲就行)
============================================================

cd d:\smart-home-esp32

git init
git add .
git commit -m "init: 项目初始化，README + 骨架代码"


echo // 刚写完 DHT11 驱动 >> src/sensors/dht_sensor.h
git add src/sensors/dht_sensor.h
git commit -m "feat: 添加温湿度传感器驱动"


echo // 发现 bug，修一下 >> src/main.cpp
git add src/main.cpp
git commit -m "fix: 修复 WiFi 连接超时未重试"


echo // 改完一版 >> src/logic/linkage_engine.h
git add .
git commit -m "feat: 完成离家/回家场景联动逻辑"


git log --oneline
# 输出:
# abc1234 feat: 完成离家/回家场景联动逻辑
# def5678 fix: 修复 WiFi 连接超时未重试
# ghi9012 feat: 添加温湿度传感器驱动
# jkl3456 init: 项目初始化，README + 骨架代码
